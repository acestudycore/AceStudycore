      delete eventSyllabusMap[id];
      if (activeEventId === id) {
        activeEventId = events[0] ? events[0].id : '';
      }
      localStorage.setItem('ace_events_v2', JSON.stringify(events));
      localStorage.setItem('ace_event_syllabi', JSON.stringify(eventSyllabusMap));
      localStorage.setItem('ace_active_evt_id', activeEventId);
      renderEvents();
      renderSubjects();
    }

    function renderEvents() {
      const container = document.getElementById('events-list');
      const selector = document.getElementById('active-event-selector');
      container.innerHTML = '';
      selector.innerHTML = '';

      if (events.length === 0) {
        container.innerHTML = `<div style="font-size:12px; color:var(--subtext);">No events created. Click + Add Timeline Event to begin.</div>`;
        document.getElementById('nearest-exam-days').innerText = '--';
        document.getElementById('nearest-exam-title').innerText = 'No Event Active';
        document.getElementById('active-event-name-display').innerText = '--';
        document.getElementById('event-subhead-name').innerText = '--';
        return;
      }

      if (!activeEventId || !events.some(e => e.id === activeEventId)) {
        activeEventId = events[0].id;
        localStorage.setItem('ace_active_evt_id', activeEventId);
      }

      const today = new Date();

      events.forEach(evt => {
        const opt = document.createElement('option');
        opt.value = evt.id;
        opt.innerText = evt.name;
        if (evt.id === activeEventId) opt.selected = true;
        selector.appendChild(opt);

        const startDate = new Date(evt.start);
        const endDate = new Date(evt.end);
        const totalDuration = endDate - startDate;
        const elapsed = today - startDate;
        let pct = 0;
        if (totalDuration > 0) {
          pct = Math.min(Math.max(Math.round((elapsed / totalDuration) * 100), 0), 100);
        }

        const isSelected = evt.id === activeEventId;

        const div = document.createElement('div');
        div.className = 'event-card';
        if (isSelected) div.style.borderColor = 'var(--accent-color)';
        div.style.cursor = 'pointer';
        div.onclick = () => switchActiveEvent(evt.id);

        div.innerHTML = `
          <div style="display:flex; justify-content:space-between; align-items:center;">
            <strong>${evt.name}</strong>
            <button class="btn-del-sm" onclick="deleteEvent('${evt.id}', event)">&times;</button>
          </div>
          <div style="font-size:11px; color:var(--subtext); margin-top:4px;">${evt.start} &rarr; ${evt.end}</div>
          <div style="font-size:11px; margin-top:6px; font-weight:bold;">${pct}% Elapsed</div>
          <div class="event-progress-bar">
            <div class="event-progress-fill" style="width: ${pct}%;"></div>
          </div>
        `;
        container.appendChild(div);
      });

      const activeEvt = events.find(e => e.id === activeEventId);
      if (activeEvt) {
        document.getElementById('active-event-name-display').innerText = activeEvt.name;
        document.getElementById('event-subhead-name').innerText = activeEvt.name;
        document.getElementById('upcoming-exam-label').innerText = activeEvt.name;

        const targetDate = new Date(activeEvt.end);
        const diffTime = targetDate - today;
        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
        const finalDays = diffDays >= 0 ? diffDays : 0;

        document.getElementById('nearest-exam-days').innerText = finalDays;
        document.getElementById('nearest-exam-title').innerText = activeEvt.name;
      }
    }

    function renderSubjects() {
      let currentSyllabus = getActiveSyllabus();
      const container = document.getElementById('subject-cards');
      container.innerHTML = '';

      let grandTotal = 0;
      let grandCovered = 0;

      for (let sub in currentSyllabus) {
        const chapters = currentSyllabus[sub];
        const subColor = getSubjectColor(sub);
        
        let doneCount = 0;
        chapters.forEach(c => {
          if (checkedChaps[`${activeEventId}-${sub}-${c}`]) doneCount++;
        });

        grandTotal += chapters.length;
        grandCovered += doneCount;

        const card = document.createElement('div');
        card.className = 'subject-card';
        card.style.borderTop = `4px solid ${subColor}`;

        let chapHtml = '';
        chapters.forEach(c => {
          const isDone = !!checkedChaps[`${activeEventId}-${sub}-${c}`];
          chapHtml += `
            <li>
              <div class="chap-left">
                <input type="checkbox" ${isDone ? 'checked' : ''} onchange="toggleChap('${sub}', '${c.replace(/'/g, "\\'")}', this.checked)">
                <span style="${isDone ? 'text-decoration: line-through; opacity: 0.6;' : ''}">${c}</span>
              </div>
              <button class="btn-del-sm" onclick="deleteChapter('${sub}', '${c.replace(/'/g, "\\'")}')">&times;</button>
            </li>
          `;
        });

        card.innerHTML = `
          <div class="subject-header">
            <div class="subject-title" style="color:${subColor};">${sub}</div>
            <div style="display:flex; align-items:center; gap:8px;">
              <span class="mini-pie-wrap">${doneCount}/${chapters.length} covered</span>
              <button class="dots-btn" onclick="openSubjectModal('${sub.replace(/'/g, "\\'")}')">&#8942;</button>
            </div>
          </div>
          <ul class="chap-list">
            ${chapHtml}
          </ul>
          <div style="margin-top: 10px;">
            <button class="btn" style="width:100%; font-size:12px; padding:4px;" onclick="addChapter('${sub.replace(/'/g, "\\'")}')">+ Add Chapter</button>
          </div>
        `;
        container.appendChild(card);
      }

      const calcPct = grandTotal > 0 ? Math.round((grandCovered / grandTotal) * 100) : 0;
      const left = grandTotal - grandCovered;

      document.getElementById('calc-total').innerText = grandTotal;
      document.getElementById('calc-completion').innerText = calcPct + '%';
      document.getElementById('calc-covered').innerText = grandCovered;
      document.getElementById('calc-left').innerText = left;

      const activeEvt = events.find(e => e.id === activeEventId);
      if (activeEvt) {
        const today = new Date();
        const targetDate = new Date(activeEvt.end);
        const diffDays = Math.max(Math.ceil((targetDate - today) / (1000 * 60 * 60 * 24)), 1);
        const pace = (left / diffDays).toFixed(1);
        document.getElementById('calc-pace').innerText = pace;
      } else {
        document.getElementById('calc-pace').innerText = '0';
      }

      document.getElementById('pie-slice').setAttribute('stroke-dasharray', `${calcPct} 100`);
    }

    // GRADE TRACKER TAB
    function addGradeScore() {
      const sub = document.getElementById('gt-sub').value.trim();
      const comp = document.getElementById('gt-comp').value.trim();
      const score = parseFloat(document.getElementById('gt-score').value);
      const total = parseFloat(document.getElementById('gt-total').value);

      if (!sub || !comp || isNaN(score) || isNaN(total) || total <= 0) return;

      const pct = Math.round((score / total) * 100);
      let grade = 'F';
      if (pct >= 90) grade = 'A*';
      else if (pct >= 80) grade = 'A';
      else if (pct >= 70) grade = 'B';
      else if (pct >= 60) grade = 'C';
      else if (pct >= 50) grade = 'D';
      else if (pct >= 40) grade = 'E';

      gradesList.push({ subject: sub, component: comp, score, total, pct, grade });
      localStorage.setItem('ace_grades', JSON.stringify(gradesList));
      renderGrades();
      renderGradeGraph();

      document.getElementById('gt-sub').value = '';
      document.getElementById('gt-comp').value = '';
      document.getElementById('gt-score').value = '';
      document.getElementById('gt-total').value = '';
    }

    function renderGradeGraph() {
      const canvas = document.getElementById('gradeTrendCanvas');
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      canvas.width = canvas.parentElement.clientWidth;
      canvas.height = canvas.parentElement.clientHeight;

      ctx.clearRect(0, 0, canvas.width, canvas.height);

      const subjects = [...new Set(gradesList.map(g => g.subject))];
      const components = [...new Set(gradesList.map(g => g.component))];

      const legend = document.getElementById('graph-legend');
      legend.innerHTML = '';

      subjects.forEach(sub => {
        const color = getSubjectColor(sub);
        legend.innerHTML += `<div class="legend-item" style="display:inline-flex; align-items:center; gap:6px; margin: 0 8px;"><div class="legend-color" style="width:12px; height:12px; border-radius:50%; background:${color}"></div><span style="font-size:13px; font-weight:bold;">${sub}</span></div>`;
      });

      if (components.length === 0) return;

      const paddingLeft = 65;
      const paddingBottom = 45;
      const paddingTop = 25;
      const paddingRight = 30;

      const chartWidth = canvas.width - paddingLeft - paddingRight;
      const chartHeight = canvas.height - paddingTop - paddingBottom;

      const textColor = getComputedStyle(document.body).getPropertyValue('--subtext').trim() || '#9aa0a6';

      // Draw Grid & Y-Axis Labels (Percentages & Grades)
      ctx.strokeStyle = '#2d3135';
      ctx.lineWidth = 1;
      ctx.fillStyle = textColor;
      ctx.font = '11px sans-serif';
      ctx.textAlign = 'right';
      ctx.textBaseline = 'middle';

      const yTicks = [
        { pct: 100, label: '100% (A*)' },
        { pct: 80, label: '80% (A)' },
        { pct: 70, label: '70% (B)' },
        { pct: 60, label: '60% (C)' },
        { pct: 50, label: '50% (D)' },
        { pct: 25, label: '25% (E)' },
        { pct: 0, label: '0% (F)' }
      ];

      yTicks.forEach(tick => {
        const y = paddingTop + (1 - tick.pct / 100) * chartHeight;
        
        ctx.beginPath();
        ctx.moveTo(paddingLeft, y);
        ctx.lineTo(canvas.width - paddingRight, y);
        ctx.stroke();

        ctx.fillText(tick.label, paddingLeft - 8, y);
      });

      // Y-Axis Title
      ctx.save();
      ctx.translate(15, canvas.height / 2);
      ctx.rotate(-Math.PI / 2);
      ctx.textAlign = 'center';
      ctx.fillStyle = getComputedStyle(document.body).getPropertyValue('--text-color').trim() || '#ffffff';
      ctx.font = 'bold 11px sans-serif';
      ctx.fillText('Grades & Percentage (%)', 0, 0);
      ctx.restore();

      // X-Axis Baseline & Labels (Exams)
      ctx.textAlign = 'center';
      ctx.textBaseline = 'top';
      components.forEach((comp, idx) => {
        const x = paddingLeft + (idx / Math.max(components.length - 1, 1)) * chartWidth;
        ctx.fillStyle = textColor;
        ctx.font = 'bold 12px sans-serif';
        ctx.fillText(comp, x, canvas.height - paddingBottom + 8);
      });

      // X-Axis Title
      ctx.fillStyle = getComputedStyle(document.body).getPropertyValue('--text-color').trim() || '#ffffff';
      ctx.font = 'bold 11px sans-serif';
      ctx.fillText('Exams / Components', paddingLeft + chartWidth / 2, canvas.height - 14);

      // Plot Subject Trend Lines
      subjects.forEach(sub => {
        const subData = gradesList.filter(g => g.subject === sub);
        const subColor = getSubjectColor(sub);

        ctx.strokeStyle = subColor;
        ctx.fillStyle = subColor;
        ctx.lineWidth = 2.5;
        ctx.beginPath();

        let started = false;
        components.forEach((comp, idx) => {
          const item = subData.find(g => g.component === comp);
          if (item) {
            const x = paddingLeft + (idx / Math.max(components.length - 1, 1)) * chartWidth;
            const y = paddingTop + (1 - item.pct / 100) * chartHeight;

            if (!started) {
              ctx.moveTo(x, y);
              started = true;
            } else {
              ctx.lineTo(x, y);
            }
          }
        });
        ctx.stroke();

        components.forEach((comp, idx) => {
          const item = subData.find(g => g.component === comp);
          if (item) {
            const x = paddingLeft + (idx / Math.max(components.length - 1, 1)) * chartWidth;
            const y = paddingTop + (1 - item.pct / 100) * chartHeight;
            ctx.beginPath();
            ctx.arc(x, y, 5, 0, Math.PI * 2);
            ctx.fill();
          }
        });
      });
    }

      ppqList.forEach((p, idx) => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td style="color:${getSubjectColor(p.sub)}; font-weight:bold;">${p.sub}</td>

        const code = codeMatch ? codeMatch[1] : p.paperCode;
        const key = `${p.sub} ${code}`;

        if (!summaryMap[key]) summaryMap[key] = { scores: [] };
        summaryMap[key].scores.push(parseFloat(p.score));
      });

      const summaryTbody = document.getElementById('ppq-summary-table-body');
      summaryTbody.innerHTML = '';

      for (let key in summaryMap) {
        const arr = summaryMap[key].scores;
        const avg = Math.round(arr.reduce((a, b) => a + b, 0) / arr.length);
        let grade = 'F';
        if (avg >= 90) grade = 'A*';
        else if (avg >= 80) grade = 'A';
        else if (avg >= 70) grade = 'B';
        else if (avg >= 60) grade = 'C';
        else if (avg >= 50) grade = 'D';

        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td><strong>${key}</strong></td>
          <td>${avg}%</td>
          <td>${avg}%</td>
          <td><span class="badge ${avg >= 75 ? 'badge-success' : avg >= 50 ? 'badge-warning' : 'badge-danger'}">${grade}</span></td>
        `;
        summaryTbody.appendChild(tr);
      }

      document.getElementById('papers-total-count').innerText = ppqList.length;

      const activeEvt = events.find(e => e.id === activeEventId);
      if (activeEvt && activeEvt.end) {
        const targetDate = new Date(activeEvt.end);
        const beforeCount = ppqList.filter(p => p.date && new Date(p.date) <= targetDate).length;
        document.getElementById('papers-before-exam-count').innerText = beforeCount;
      } else {
        document.getElementById('papers-before-exam-count').innerText = ppqList.length;
      }

      topicalList.push({ sub, topic, done, total });
      localStorage.setItem('ace_topical', JSON.stringify(topicalList));
      renderBooklets();

      document.getElementById('top-sub').value = '';
      document.getElementById('top-topic').value = '';
      document.getElementById('top-done').value = '';
      document.getElementById('top-total').value = '';
      toggleTopicalForm();
    }

    function adjustWbPages(idx, delta) {
      if (workbookList[idx]) {
        workbookList[idx].done = Math.min(Math.max(0, workbookList[idx].done + delta), workbookList[idx].total);
        localStorage.setItem('ace_workbook', JSON.stringify(workbookList));
        renderBooklets();
      }
    }

    function updateWbPagesInput(idx, val) {
      if (workbookList[idx]) {
        let parsed = parseInt(val) || 0;
        workbookList[idx].done = Math.min(Math.max(0, parsed), workbookList[idx].total);
        localStorage.setItem('ace_workbook', JSON.stringify(workbookList));
        renderBooklets();
      }
    }

    function adjustTopPages(idx, delta) {
      if (topicalList[idx]) {
        topicalList[idx].done = Math.min(Math.max(0, topicalList[idx].done + delta), topicalList[idx].total);
        localStorage.setItem('ace_topical', JSON.stringify(topicalList));
        renderBooklets();
      }
    }

    function updateTopPagesInput(idx, val) {
      if (topicalList[idx]) {
        let parsed = parseInt(val) || 0;
        topicalList[idx].done = Math.min(Math.max(0, parsed), topicalList[idx].total);

        renderBooklets();

    }

    function deleteWorkbook(idx) {
      workbookList.splice(idx, 1);
      localStorage.setItem('ace_workbook', JSON.stringify(workbookList));

    }

      topicalList.splice(idx, 1);
      localStorage.setItem('ace_topical', JSON.stringify(topicalList));
      renderBooklets();
    }

    // POMODORO TIMER
    let pomoSeconds = 25 * 60;
    let pomoInterval = null;

    function updatePomoDisplay() {
      const m = Math.floor(pomoSeconds / 60).toString().padStart(2, '0');
      const s = (pomoSeconds % 60).toString().padStart(2, '0');
      document.getElementById('pomo-display').innerText = `${m}:${s}`;
    }

    function applyCustomPomo() {
      const val = parseInt(document.getElementById('pomo-input-min').value);
      if (val && val > 0) {
        pausePomo();
        pomoSeconds = val * 60;
        updatePomoDisplay();
      }
    }

    function startPomo() {
      if (pomoInterval) return;
      pomoInterval = setInterval(() => {
        if (pomoSeconds > 0) {
          pomoSeconds--;
          updatePomoDisplay();
        } else {
          pausePomo();
          alert("Focus time completed!");
        }
      }, 1000);
    }

    function pausePomo() {
      clearInterval(pomoInterval);
      pomoInterval = null;
    }

    function resetPomo() {
      pausePomo();
      const val = parseInt(document.getElementById('pomo-input-min').value) || 25;
      pomoSeconds = val * 60;
      updatePomoDisplay();
    }
